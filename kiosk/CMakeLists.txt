cmake_minimum_required(VERSION 3.16)
project(ScanStationUI C CXX ASM)
set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
set(LV_CONF_BUILD_DISABLE_EXAMPLES ON)
set(LV_CONF_BUILD_DISABLE_DEMOS ON)
set(LV_CONF_BUILD_DISABLE_THORVG_INTERNAL ON)
add_subdirectory("${LVGL_SOURCE}" lvgl EXCLUDE_FROM_ALL)
add_executable(kiosk main.c)
target_link_libraries(kiosk PRIVATE lvgl m)
target_compile_options(kiosk PRIVATE -Wall -Wextra)
